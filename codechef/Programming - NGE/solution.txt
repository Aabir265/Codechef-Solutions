#include <bits/stdc++.h>
using namespace std;

int main() {
  
    int n;
    if (!(cin >> n)) return 0;

    vector<long long> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    vector<long long> nge(n, -1);
    stack<long long> st;

    // Traverse the array from right to left
    for (int i = n - 1; i >= 0; i--) {
        // Pop elements from stack that are smaller than or equal to current element
        while (!st.empty() && st.top() <= a[i]) {
            st.pop();
        }

        // If stack is not empty, top element is the next greater element
        if (!st.empty()) {
            nge[i] = st.top();
        }

        // Push current element onto stack
        st.push(a[i]);
    }

    // Print results
    for (int i = 0; i < n; i++) {
        cout << nge[i] << (i == n - 1 ? "" : " ");
    }
    cout << "
";

    return 0;
}
