#include <bits/stdc++.h>
using namespace std;

struct Node {
    int val;
    Node* left;
    Node* right;

    Node(int val) {
        this->val = val;
        left = right = NULL;
    }
};

class Solution {
  public:
    
    int height(Node* root) {
        if (root == NULL) {
            return 0;
        }

        return 1 + max(height(root->left),
                       height(root->right));
    }

    bool checkBalanced(Node* root) {
        if (root == NULL) {
            return true;
        }

        int leftHeight = height(root->left);
        int rightHeight = height(root->right);

        if (abs(leftHeight - rightHeight) > 1) {
            return false;
        }

        return checkBalanced(root->left) &&
               checkBalanced(root->right);
    }

    void isHeightBalanced(Node* root) {
        if (checkBalanced(root)) {
            cout << "YES" << '
';
        } else {
            cout << "NO" << '
';
        }
    }
};


void inOrder(Node* root, vector<int> &nodes){
    if(root == NULL)
      return;
    inOrder(root -> left, nodes);
    nodes.push_back(root->val);
    inOrder(root -> right, nodes);
}
Node* buildBinaryTree(vector<pair<pair<int, int>, char>>& edges) {
    map<int, Node*> nodes;
    set<Node*> children;
    for (const auto& edge : edges) {
        int parentVal = edge.first.first;
        int childVal = edge.first.second;
        char direction = edge.second;
        Node* parentNode;
        Node* childNode;
        if (nodes.find(parentVal) == nodes.end()) {
            parentNode = new Node(parentVal);
            nodes[parentVal] = parentNode;
        } else {
            parentNode = nodes[parentVal];
        }
        if(nodes.find(childVal) == nodes.end()) {
            childNode = new Node(childVal);
            nodes[childVal] = childNode;
        } else {
            childNode = nodes[childVal];
        }
        if (direction == 'L') {
            parentNode->left = childNode;
        } else {
            parentNode->right = childNode;
        }
        children.insert(childNode);
    }
    Node* root = NULL;
    for (const auto& it : nodes) {
        if (children.find(it.second) == children.end()) {
            root = it.second;
            break;
        }
    }
    return root;
}
