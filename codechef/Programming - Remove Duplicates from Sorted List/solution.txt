#include <bits/stdc++.h>
using namespace std;

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    cin >> T;

    while (T--) {
        int N;
        cin >> N;

        int prev;
        cin >> prev;
        cout << prev;

        for (int i = 1; i < N; i++) {
            int x;
            cin >> x;

            if (x != prev) {
                cout << ' ' << x;
                prev = x;
            }
        }

        cout << '
';
    }

    return 0;
}
